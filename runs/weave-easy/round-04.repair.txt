def _parse(s):
    parts = s.split('_')
    m = {}
    for i, row in enumerate(parts):
        if len(row) < 6: continue
        for r in (0, 1):
            tok = row[3*r:3*r+3]
            if tok == '---': continue
            w, mv, ns = tok[0], tok[1], tok[2]
            m[chr(ord('A')+i, r, w)] = (ns, 1 if mv=='R' else -1)
    return m
